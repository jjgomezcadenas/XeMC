\section{Physics implemented}

\subsection{Photon interactions (from NIST XCOM tables)}

\begin{itemize}
  \item \textbf{Compton scattering}: Klein-Nishina sampled by composition+rejection (Butcher-Messel). Scattered photon continues; recoil electron pushed onto stack.
  \item \textbf{Pair production}: Bethe-Heitler energy split with flat+rejection sampling. Positron and electron pushed onto stack. Positrons annihilate at rest (two 511\,keV back-to-back photons).
  \item \textbf{Photoelectric absorption}: above K-edge, $E_K$ deposited locally (relaxation cascade is sub-mm); photoelectron tracked if $T_e > T_{e,\mathrm{cut}}$. Below K-edge, full energy deposited locally.
\end{itemize}

\subsection{Electron/positron transport}

\begin{itemize}
  \item Condensed-history with fixed step size (1\,mm).
  \item Continuous collisional energy loss from NIST ESTAR tables.
  \item Discrete hard bremsstrahlung ($k > k_{\min}$) via $1/k$ envelope rejection with pre-tabulated cross section and rejection ceiling.
  \item No multiple scattering (straight-line steps) --- adequate for LXe where electron range $\sim$\,mm.
  \item No energy-loss straggling.
\end{itemize}

\subsection{What is NOT implemented}

\begin{itemize}
  \item Coherent (Rayleigh) scattering --- negligible above 100\,keV.
  \item Multiple scattering --- needed for gas, not for LXe at current resolution.
  \item Delta-ray production.
  \item Energy-loss straggling (Landau/Vavilov fluctuations).
  \item Source generation (position, direction, spectrum sampling).
  \item Multi-volume transport (particle crossing volume boundaries).
  \item Passive shielding attenuation.
\end{itemize}

\section{Geometry and materials}

\subsection{Geometry hierarchy}

\begin{verbatim}
Cyl / Box          (geometric solid: dimensions)
  +-- LCyl / LBox   (logical volume: solid + position in MARS)
      +-- PCyl / PBox  (physical volume: logical + material)
          +-- RCyl      (radioactive volume: physical + specific activities)

Detector = MARS (vacuum) + list of physical volumes
\end{verbatim}

\subsection{Defined detector: LZ}

\begin{itemize}
  \item MARS: cylinder $R = 110$\,cm, $H_{\mathrm{half}} = 155$\,cm, Vacuum
  \item LXeTPC: cylinder $R = 72.8$\,cm, $H_{\mathrm{half}} = 72.8$\,cm, LXe, centered at origin, mass $\sim$7160\,kg
\end{itemize}

\subsection{Material capability flags}

\begin{itemize}
  \item \textbf{active}: XCOM + ESTAR + brems tables loaded. Can do photon + electron transport.
  \item \textbf{tracking}: condensed-history electron stepping (straight-line, no MS).
  \item \textbf{full\_tracking} (future): + multiple scattering + straggling for gas.
\end{itemize}

\section{Source generation}

\textbf{Current limitation}: primary photon is generated at a fixed position (default origin) with fixed direction (default $+z$) and fixed energy (caller-specified). No built-in source sampling.

\textbf{Needed}:
\begin{itemize}
  \item Uniform generation within a volume (for bulk contamination).
  \item Surface generation (for external backgrounds).
  \item Isotropic direction sampling.
  \item Spectrum sampling (fixed gamma lines: 2.448\,MeV Bi-214, 2.615\,MeV Tl-208).
  \item Integration with \texttt{RCyl} radioactive volumes for flux-weighted generation.
\end{itemize}

\section{Output and analysis}

\begin{itemize}
  \item \textbf{Deposits}: list of (position, energy, source\_label) per event.
  \item \textbf{Clustering}: \texttt{cluster\_deposits\_in\_z(deposits, dz\_cm; E\_min)} groups deposits by $z$-proximity. Energy-weighted centroid. Sub-threshold clusters discarded.
  \item \textbf{SS/MS classification}: \texttt{is\_single\_site(deposits, dz, E\_min)} --- true if $\leq 1$ cluster survives.
  \item \textbf{Driver script} (\texttt{scripts/run\_test.jl}): CLI with \texttt{-{}-nevents}, \texttt{-{}-energy}, \texttt{-{}-mode}, \texttt{-{}-detector}, \texttt{-{}-save-events}. Outputs summary to stdout + \texttt{summary.txt}; optionally writes long-format \texttt{events.csv} (one row per cluster, event fields repeated).
\end{itemize}
